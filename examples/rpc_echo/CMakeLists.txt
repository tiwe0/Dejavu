cmake_minimum_required(VERSION 3.28)
project(dejavu_rpc_echo LANGUAGES CXX)
if(NOT ANDROID OR NOT ANDROID_ABI STREQUAL "arm64-v8a")
    message(FATAL_ERROR "RPC echo requires Android arm64-v8a")
endif()
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(DEJAVU_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_library(dejavu_rpc_echo SHARED plugin_entry.cpp)
target_include_directories(dejavu_rpc_echo PRIVATE ${DEJAVU_ROOT}/include)
target_compile_options(dejavu_rpc_echo PRIVATE -Wall -Wextra -fvisibility=hidden)
target_link_options(dejavu_rpc_echo PRIVATE -Wl,--no-undefined -Wl,-z,max-page-size=16384)

set_property(TARGET dejavu_rpc_echo APPEND PROPERTY LINK_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/exports.map")
target_link_options(dejavu_rpc_echo PRIVATE "-Wl,--version-script=${CMAKE_CURRENT_SOURCE_DIR}/exports.map")
